// === Настройки ===
#define UART_PORT UART_NUM_1        // Используем UART1
#define UART_TX_PIN GPIO_NUM_4      // TX на GPIO4
#define UART_RX_PIN GPIO_NUM_5      // RX на GPIO5
#define BAUD_RATE 115200            // Скорость передачи
#define TX_BUFFER_SIZE 6500         // Размер TX буфера
#define RX_BUFFER_SIZE 2048          // Размер RX буфера

static const char *TAG = "UART_ECHO";

// Основная функция
void app_main(void) {
    // === Конфигурация UART1 ===
    uart_config_t uart_config = {
        .baud_rate = BAUD_RATE,
        .data_bits = UART_DATA_8_BITS,
        .parity = UART_PARITY_DISABLE,
        .stop_bits = UART_STOP_BITS_1,
        .flow_ctrl = UART_HW_FLOWCTRL_DISABLE,
    };

    // Устанавливаем параметры UART
    uart_param_config(UART_PORT, &uart_config);

    // Устанавливаем пины UART1
    uart_set_pin(UART_PORT, UART_TX_PIN, UART_RX_PIN, UART_PIN_NO_CHANGE, UART_PIN_NO_CHANGE);

    // Устанавливаем драйвер UART1
    esp_err_t err = uart_driver_install(UART_PORT, TX_BUFFER_SIZE, RX_BUFFER_SIZE, 0, NULL, 0);
    if (err != ESP_OK) {
        ESP_LOGE(TAG, "Ошибка установки драйвера UART: %x", err);
        return;
    }

    // Буфер для чтения
    uint8_t rx_buffer[RX_BUFFER_SIZE];
        const char *hello_msg = "Hello from ESP32-C6!\r\n";
        uart_write_bytes(UART_PORT, hello_msg, strlen(hello_msg));

        ESP_LOGI(TAG, "Sent: %s", hello_msg);

    while (1) {
        // === Периодическая отправка данных в UART ===
        

        // === Чтение входящих данных ===
        int len = uart_read_bytes(UART_PORT, rx_buffer, RX_BUFFER_SIZE - 1, pdMS_TO_TICKS(100));
        if (len > 0) {
            rx_buffer[len] = '\0';
            ESP_LOGI(TAG, "Получено %d байт: %s", len, rx_buffer);

            // echo
            uart_write_bytes(UART_PORT, "%d", len);
            uart_write_bytes(UART_PORT, "Received: ", len);
            uart_write_bytes(UART_PORT, (const char *)rx_buffer, len);
            uart_write_bytes(UART_PORT, "\r\n", 2);
        }
        // Пауза 2 секунды
        // vTaskDelay(pdMS_TO_TICKS(2000));
    }
}

// Обработчик прерывания
static void gpio_isr_handler(void* arg) {
    uint32_t gpio_num = (uint32_t) arg;
    uint64_t current_time = esp_timer_get_time(); // Получаем текущее время в микросекундах
    uint64_t curr_pin_diff = (current_time - last_press_time);   //время между 2 последними "гвоздями"
    if (curr_pin_diff < 700) {
        return;
    }


    if (curr_pin_diff > (last_pin_diff * 2)) { // если сейчас промежуток сильно больше, чем прошлый
        pinCount = 0;
        rev_time = current_time - last_rev_time; //время оборота = текущее время - предыдущий момент оборота
        changed = 1;                             //выставляем флаг
        last_rev_time = current_time;            //выставляем предыдущий момент оборота

        //здесь нужно запускать управление свечой
    } else {
        pinCount++;
    } 

    last_pin_diff = curr_pin_diff;             //обновляем промежуток для следующего раза
    
    last_press_time = current_time;              //обновляем время последнего срабатывания
}
